\subsubsection{Löse die folgenden Gleichungen:} % I.

\begin{enumerate}

  % 1.
  \item Löse die Gleichung \(2x + 4 = 18\).
  \[
  \begin{aligned}
    2x + 4 &= 18 \quad &\big| - 4 \\
    2x &= 14 \quad &\big| : 2 \\
    x &= \mathbf{7} \\
    2 \cdot 7 + 4 = 14 + 4 &= 18
  \end{aligned}
  \]

  % 2.
  \item Löse die Gleichung \(6x + 3 = 33\).
  \[
  \begin{aligned}
    6x + 3 &= 33 &\quad\big| - 3 \\
    6x &= 30 &\quad\big| : 6 \\
    x &= \mathbf{5} \\
    6 \cdot 5 + 3 = 30 + 3 &= 33
  \end{aligned}
  \]

  % 3.
  \item Löse die Gleichung \(5x + 8 = 13\).
  \[
  \begin{aligned}
    5x + 8 &= 13 &\quad\big| - 8 \\
    5x &= 5 &\quad\big| : 5 \\
    x &= \mathbf{1} \\
    5 \cdot 1 + 8 = 5 + 8 &= 13
  \end{aligned}
  \]

  % 4.
  \item Löse die Gleichung \(3x + 9 = 18\).
  \[
  \begin{aligned}
    3x + 9 &= 18 &\quad\big| - 9 \\
    3x &= 9 &\quad\big| : 3 \\
    x &= \mathbf{3} \\
    3 \cdot 3 + 9 = 9 + 9 &= 18
  \end{aligned}
  \]

  % 5.
  \item Löse die Gleichung \(6x + 9 = 45\).
  \[
  \begin{aligned}
    6x + 9 &= 45 &\quad\big| - 9 \\
    6x &= 36 &\quad\big| : 6 \\
    x &= \mathbf{6} \\
    6 \cdot 6 + 9 = 36 + 9 &= 45
  \end{aligned}
  \]

  % 6.
  \item Löse die Gleichung \(4x + 9 = 29\).
  \[
  \begin{aligned}
    4x + 9 &= 29 &\quad\big| - 9 \\
    4x &= 20 &\quad\big| : 4 \\
    x &= \mathbf{5} \\
    4 \cdot 5 + 9 = 20 + 9 &= 29
  \end{aligned}
  \]

  % 7.
  \item Löse die Gleichung \(7x + 4 = 25\).
  \[
  \begin{aligned}
    7x + 4 &= 25 &\quad\big| - 4 \\
    7x &= 21 &\quad\big| : 7 \\
    x &= \mathbf{3} \\
    7 \cdot 3 + 4 = 21 + 4 &= 25
  \end{aligned}
  \]

  % 8.
  \item Löse die Gleichung \(2x + 7 = 11\).
  \[
  \begin{aligned}
    2x + 7 &= 11 &\quad\big| - 7 \\
    2x &= 4 &\quad\big| : 2 \\
    x &= \mathbf{2} \\
    2 \cdot 2 + 7 = 4 + 7 &= 11
  \end{aligned}
  \]

  % 9.
  \item Löse die Gleichung \(4x + 8 = 36\).
  \[
  \begin{aligned}
    4x + 8 &= 36 &\quad\big| - 8 \\
    4x &= 28 &\quad\big| : 4 \\
    x &= \mathbf{7} \\
    4 \cdot 7 + 8 = 28 + 8 &= 36
  \end{aligned}
  \]

  % 10.
  \item Löse die Gleichung \(6x + 4 = 46\).
  \[
  \begin{aligned}
    6x + 4 &= 46 &\quad\big| - 4 \\
    6x &= 42 &\quad\big| : 6 \\
    x &= \mathbf{7} \\
    6 \cdot 7 + 4 = 42 + 4 &= 46
  \end{aligned}
  \]

\end{enumerate}

      \cfrac{7x + 1}{x + 1} &= 4 &\quad\big| \cdot (x + 1) \\
      7x + 1 &= 4x + 4 &\quad\big| - 4x - 1 \\
      3x &= 3 &\quad\big| : 3 \\
      x &= \mathbf{1}
    \end{aligned}
  \]

\end{enumerate}

\subsubsection{Löse die folgenden Bruchgleichungen:}

\begin{enumerate}[label=\alph*)]

  % a)
  \item \(\cfrac{x + 6}{x + 2} = \cfrac{x + 4}{x + 1}\)
  \[
    \begin{aligned}
      \cfrac{x + 6}{x + 2} &= \cfrac{x + 4}{x + 1} &\quad\big| \cdot (x + 2) \cdot (x + 1) \\
      (x + 6) \cdot (x + 1) &= (x + 4) \cdot (x + 2) \\
      x^2 + 7x + 6 &= x^2 + 6x + 8 &\quad\big| - x^2 - 6x - 6 \\
      x &= \mathbf{2}
    \end{aligned}
  \]

  % b)
  \item \(\cfrac{x + 5}{x + 1} = \cfrac{x + 7}{x + 2}\)
  \[
    \begin{aligned}
      \cfrac{x + 5}{x + 1} &= \cfrac{x + 7}{x + 2} &\quad\big| \cdot (x + 1) \cdot (x + 2) \\
      (x + 5) \cdot (x + 2) &= (x + 7) \cdot (x + 1) \\
      x^2 + 7x + 10 &= x^2 + 8x + 7 &\quad\big| - x^2 - 8x - 10 \\
      -x &= -3 &\quad\big| : (-1) \\
      x &= \mathbf{3}
    \end{aligned}
  \]

  % c)
  \item \(\cfrac{x + 1}{3x + 1} = \cfrac{x + 2}{3x + 3}\)
  \[
    \begin{aligned}
      \cfrac{x + 1}{3x + 1} &= \cfrac{x + 2}{3x + 3} &\quad\big| \cdot (3x + 1) \cdot (3x + 3) \\
      (x + 1) \cdot (3x + 3) &= (x + 2) \cdot (3x + 1) \\
      3x^2 + 6x + 3 &= 3x^2 + 7x + 2 &\quad\big| - 3x^2 - 7x - 3 \\
      -x &= -1 &\quad\big| : (-1) \\
      x &= \mathbf{1}
    \end{aligned}
  \]

  % d)
  \item \(\cfrac{5x}{x + 1} = \cfrac{5x - 8}{x - 1}\)
  \[
    \begin{aligned}
      \cfrac{5x}{x + 1} &= \cfrac{5x - 8}{x - 1} &\quad\big| \cdot (x + 1) \cdot (x - 1) \\
      5x \cdot (x - 1) &= (5x - 8) \cdot (x + 1) \\
      5x^2 - 5x &= 5x^2 - 3x - 8 &\quad\big| - 5x^2 + 3x \\
      -2x &= -8 &\quad\big| : (-2) \\
      x &= \mathbf{4}
    \end{aligned}
  \]

\end{enumerate}

\subsubsection{Löse die folgenden Bruchgleichungen (d.h. finde x):}

\begin{enumerate}[label=\alph*)]

  % a)
  \item \(ax + b = c\)
  \[
    \begin{aligned}
      ax + b &= c &\quad\big| - b \\
      ax &= c - b &\quad\big| : a \\
      x &= \mathbf{\cfrac{c - b}{a}}
    \end{aligned}
  \]

  % b)
  \item \(\cfrac{a}{x + b} = c\)
  \[
    \begin{aligned}
      \cfrac{a}{x + b} &= c &\quad\big| \cdot (x + b) \\
      a = c \cdot (x + b) &= cx + bc &\quad\big| - bc \\
      a - bc &= cx &\quad\big| : c \\
      x &= \mathbf{\cfrac{a - bc}{c}}
    \end{aligned}
  \]

  % c)
  \item \(\cfrac{c}{ax + b} = d\)
  \[
    \begin{aligned}
      \cfrac{c}{ax + b} &= d &\quad\big| \cdot (ax + b) \\
      c = d \cdot (ax + b) &= adx + bd &\quad\big| - bd \\
      c - bd &= adx &\quad\big| : ad \\
      x &= \mathbf{\cfrac{c - bd}{ad}}
    \end{aligned}
  \]

  % d)
  \item \(\cfrac{ax + b}{cx + d} = e\)
  \[
    \begin{aligned}
      \cfrac{ax + b}{cx + d} &= e &\quad\big| \cdot (cx + d) \\
      ax + b = e \cdot (cx + d) &= cex + de &\quad\big| - cex - b \\
      ax - cex = x \cdot (a - ce) &= de - b &\quad\big| : (a - ce) \\
      x &= \mathbf{\cfrac{de - b}{a - ce}} = \mathbf{\cfrac{b - de}{cd - a}}
    \end{aligned}
  \]

  % e)
  \item \(\cfrac{c}{ax + b} = \cfrac{f}{dx + e}\)
  \[
    \begin{aligned}
      \cfrac{c}{ax + b} &= \cfrac{f}{dx + e} &\quad\big| \cdot (ax + b) \cdot (dx + e) \\
      c \cdot (dx + e) &= f \cdot (ax + b) \\
      cdx + ce &= afx + bf &\quad\big| - afx - ce \\
      cdx - afx = x \cdot (cd - af) &= bf -ce &\quad\big| : (cd - af) \\
      x &= \mathbf{\cfrac{bf - ce}{cd - af}} = \mathbf{\cfrac{ce - bf}{af - cd}}
    \end{aligned}
  \]

\end{enumerate}

\section{Anwendungen}

\subsection{Aufgaben zu Kapitel 5.1 \\
  - Textaufgaben}

\begin{enumerate}

  % 1.
  \item Eine dreiköpfige Familie mit einem minderjährigen Kind geht in ein Kino.
    Ein Erwachsenenticket kostet doppelt so viel wie ein Kinderticket.
    Die Familie zahlt insgesamt \(32,50\text{\euro}\). Wie viel kostet ein Kinderticket?

    Ein Erwachsenenticket kostet demnach doppelt soviel wie ein Kinderticket.
    Die Gleichung lautet also:

    \[
      \begin{aligned}
        2x \cdot (2 + \cfrac{1}{2}) &= \\
        4x + x &= \\
        5x &= 32,50\text{\euro} &\quad\big| : 5 \\
        x &= \mathbf{6,50\text{\euro}}
      \end{aligned}
    \]

    Ein Kinderticket kostet also \(\mathbf{6,50\text{\euro}}\).

  % 2.
  \item Eine vierköpfige Familie mit zwei minderjährigen Kindern besucht einen Zoo.
    Ein Kinderticket kostet \(4,00\text{\euro}\) weniger als ein Erwachsenenticket.
    Die Familie zahlt insgesamt \(43,50\text{\euro}\). Was kostet ein Erwachsenenticket?

    Ein Erwachsenenticket kostet demnach \(x\) und ein Kinderticket \(x - 4\text{\euro}\).
    Die Gleichung lautet also:

    \[
      \begin{aligned}
        2x + 2 \cdot (x - 4,00\text{\euro}) &= \\
        2x + 2x - 8,00\text{\euro} &= \\
        4x - 8,00\text{\euro} &= 43,50\text{\euro} &\quad\big| + 8,00\text{\euro} \\
        4x &= 51,50\text{\euro} &\quad\big| : 4 \\
        x &= \mathbf{12,875\text{\euro}} \approx \mathbf{12,90\text{\euro}}
      \end{aligned}
    \]

    Ein Erwachsenenticket kostet also \(\mathbf{12,90\text{\euro}}\).

  % 3.
  \item Frau Müller kauft \(3,2kg\) Bananen und Herr Schmidt \(1,7kg\).
    Frau Müller zahlt \(2,40\text{\euro}\) mehr als Herr Schmidt.
    Wie viel kostet Kilogramm Bananen?

    Die Differenz zwischen den Preisen beträgt \(2,40\text{\euro}\).
    Die Differenz in der Menge beträgt \(3,2kg - 1,7kg = 1,5kg\).
    Der Preis pro Kilogramm ist also:

    \[
      \begin{aligned}
        \cfrac{2,40\text{\euro}}{1,5\text{kg}} &= \mathbf{1,60\text{\euro}/\text{kg}}
      \end{aligned}
    \]

    Ein Kilogramm Bananen kostet also \(\mathbf{1,60\text{\euro}}\).

  % 4.
  \item Drei Artikel A, B und C kosten zusammen \(23,80\text{\euro}\). Der Artikel B ist
    dreimal so teuer wie Artikel A. Der Artikel C kostet \(4,00\text{\euro}\) mehr als Artikel A.
    Was kosten die drei Artikel? (Tipp: Berechne zuerst den Preis von A.)

    \[
      \begin{aligned}
        A + B + C &= 23,80\text{\euro} \\
        B &= 3 \cdot A \\
        C &= A + 4,00\text{\euro} \\
        A + 3 \cdot A + A + 4,00\text{\euro} &= 23,80\text{\euro} &\quad\big| - 4,00\text{\euro} \\
        5 \cdot A &= 19,80\text{\euro} &\quad\big| : 5 \\
        A &= \mathbf{3,96\text{\euro}} \\
        B = 3 \cdot 3,96\text{\euro} &= \mathbf{11,88\text{\euro}} \\
        C = 3,96\text{\euro} + 4,00\text{\euro} &= \mathbf{7,96\text{\euro}}
      \end{aligned}
    \]

    Die Artikel kosten also \(A = \mathbf{3,96\text{\euro}}\), \(B = \mathbf{11,88\text{\euro}}\)
    und \(\mathbf{C = 7,96\text{\euro}}\).

  % 5.
  \item Die Geschwister Lisa und Thomas sind zusammen 31 Jahre alt.
    Lisa ist 5 Jahre älter als Thomas. Wie alt ist Thomas? Wie alt ist Lisa?

    \[
      \begin{aligned}
        Lisa + Thomas &= 31 \\
        Lisa &= Thomas + 5 \\
        Thomas + 5 + Thomas &= 31 &\quad\big| - 5 \\
        2 \cdot Thomas &= 26 &\quad\big| : 2 \\
        Thomas &= \mathbf{13} \\
        Lisa = 13 + 5 &= \mathbf{18}
      \end{aligned}
    \]

    Lisa ist \textbf{18} und Thomas \textbf{13} Jahre alt.

  % 6.
  \item In 30 Jahren wird eine gewisse Person 11 Jahre älter als doppelt so alt wie jetzt sein.
    Wie alt ist die Person jetzt?

    \[
      \begin{aligned}
        x + 30 &= 2x + 11 &\quad\big| - 2x - 30 \\
        -x &= -19 &\quad\big| : (-1) \\
        x &= \mathbf{19}
      \end{aligned}
    \]

    Die Person ist also jetzt \(\mathbf{19}\) Jahre alt.

  % 7.
  \item Markus ist dreimal so alt wie Benjamin. In 4 Jahren wird Markus nur noch
    doppelt so alt wie Benjamin sein. Wie alt ist Benjamin? Wie alt ist Markus?

    \[
      \begin{aligned}
        Markus &= 3 \cdot Benjamin \\
        Markus + 4 = 2 \cdot (Benjamin + 4) &= 2 \cdot Benjamin + 8 &\quad\big| - 4 \\
        Markus &= 2 \cdot Benjamin + 4 &\quad\big| - 2 \cdot Benjamin \\
        Markus - 2 \cdot Benjamin &= 4 &\quad\big| : 1 \\
        3 \cdot Benjamin - 2 \cdot Benjamin &= 4 \\
        Benjamin &= \mathbf{4} \\
        Markus = 3 \cdot 4 &= \mathbf{12}
      \end{aligned}
    \]

    Benjamin ist \textbf{4} und Markus \textbf{12} Jahre alt.

  % 8.
  \item Bei einem rechteckigen Grundstück unterscheiden sich die Länge und Breite
    um \(9m\). Der Zaun um das Grundstück misst \(102m\). Wie breit ist das Grundstück?

    \[
      \begin{aligned}
        2 \cdot (\textnormal{Länge} + Breite) &= 102 &\quad\big| : 2 \\
        \textnormal{Länge} + Breite &= 51 \\
        \textnormal{Länge} = Breite + 9 \\
        Breite + 9 + Breite &= 51 &\quad\big| - 9 \\
        2 \cdot Breite &= 42 &\quad\big| : 2 \\
        Breite &= \mathbf{21}
      \end{aligned}
    \]

    Das Grundstück ist also \(\mathbf{21m}\) breit.

  % 9.
  \item Bei einem Dreieck ist eine Seite \(4cm\) länger als die kürzeste Seite.
    Die andere Seite ist dreimal so lang wie die kürzeste. Der Umfang beträgt 19cm.
    Wie lange sind die Seiten des Dreiecks?

    \[
      \begin{aligned}
        \textnormal{Seite}_1 + \textnormal{Seite}_2 + \textnormal{Seite}_3 &= 19cm \\
        \textnormal{Seite}_1 &= x \\
        \textnormal{Seite}_2 &= x + 4cm \\
        \textnormal{Seite}_3 &= 3x \\
        x + x + 4cm + 3x = 5x + 4cm &= 19cm &\quad\big| - 4cm \\
        5x &= 15cm &\quad\big| : 5 \\
        x &= \mathbf{3cm} \\
        \textnormal{Seite}_1 &= \mathbf{3cm} \\
        \textnormal{Seite}_2 = 3cm + 4cm &= \mathbf{7cm} \\
        \textnormal{Seite}_3 = 3 \cdot 3cm &= \mathbf{9cm}
      \end{aligned}
    \]

    Die Seiten des Dreiecks sind also \(\mathbf{3cm}\), \(\mathbf{7cm}\) und \(\mathbf{9cm}\) lang.

  % 10.
  \item Der Flächeninhalt eines Quadrates nimmt um \(57cm^2\) zu,
    wenn die Seiten um \(3cm\) verlängert werden.
    Wie lange sind die Seiten des ursprünglichen Quadrates?

    \[
      \begin{aligned}
        (x + 3cm)^2 - x^2 &= 57cm^2 \\
        x^2 + 6cm \cdot x + 9cm^2 - x^2 &= 57cm^2 &\quad\big| - 9cm^2 \\
        6cm \cdot x &= 48cm^2 &\quad\big| : 6cm \\
        x &= \mathbf{8cm}
      \end{aligned}
    \]

    Die Seite des Quadrates ist also \(\mathbf{8cm}\) lang.

  % 11.
  \item Herr Jansen tankt \(32l\) Benzin und zahlt dafür \(10,35\text{\euro}\)
    weniger als Herr Bleifuß, der an derselben Tankstelle \(41l\) des gleichen Benzins tank.
    Wie viel kostet ein Liter Benzin?

    \[
      \begin{aligned}
        41l - 32l &= 9l \\
        \cfrac{10,35\text{\euro}}{9l} &= \mathbf{1,15\text{\euro}/l}
      \end{aligned}
    \]

    Ein Liter Benzin kostet also \(\mathbf{1,15\text{\euro}}\).

  % 12.
  \item Wenn man die Seiten eines Quadrates um \(2,5cm\) verlängert, so wird der Umfang
    1,5-mal so lang. Wie lange sind die Seiten des ursprünglichen Quadrates?

    \[
      \begin{aligned}
        U_1 &= 4 \cdot x \\
        U_2 &= U_1 \cdot 1,5 = 1,5 \cdot 4 \cdot x = 6 \cdot x \\
        U_2 &= 4 \cdot (x + 2,5cm) \\
        6 \cdot x &= 4 \cdot (x + 2,5cm) &\quad\big| - 4x \\
        2 \cdot x &= 10cm &\quad\big| : 2 \\
        x &= \mathbf{5cm}
      \end{aligned}
    \]

    Die Seite des Quadrates ist also \(\mathbf{5cm}\) lang.

  % 13.
  \item Wenn man die Seiten eines Quadrates um \(3cm\) verlängert, so erhöht sich
    der Flächininhalt um \(39cm^2\). Wie lang sind die Seiten des ursprünglichen Quadrates?

    \[
      \begin{aligned}
        A_1 &= x^2 \\
        A_2 &= A_1 + 39cm^2 = x^2 + 39cm^2 \\
        A_2 &= (x + 3cm)^2 \\
        (x + 3cm)^2 &= x^2 + 39cm^2 \\
        x^2 + 6cm \cdot x + 9cm^2 &= x^2 + 39cm^2 &\quad\big| - x^2 - 9cm^2 \\
        6cm \cdot x &= 30cm^2 &\quad\big| : (6cm) \\
        x &= \mathbf{5cm}
      \end{aligned}
    \]

    Die Seite des Quadrates ist also \(\mathbf{5cm}\) lang.

  % 14.
  \item Wenn man die Seiten eines Quadrates um \(1,8m\) verringert, so beträgt der
    Umfang nur noch zwei Drittel des ursprünglichen.
    Wie lange sind die Seiten des ursprünglichen Quadrates?

    \[
      \begin{aligned}
        U_1 &= 4 \cdot x \\
        U_2 &= U_1 \cdot \cfrac{2}{3} = \cfrac{2}{3} \cdot 4 \cdot x = \cfrac{8}{3} \cdot x \\
        U_2 &= 4 \cdot (x - 1,8m) \\
        \cfrac{8}{3} \cdot x = 4 \cdot (x - 1,8m) &= 4 \cdot x - 7,2m &\quad\big| - 4x \\
        \cfrac{8}{3} \cdot x - 4x &= -7,2m &\quad\big| \cdot 3 \\
        8x - 12x = -4x &= -21,6m &\quad\big| : (-4) \\
        x &= \mathbf{5,4m}
      \end{aligned}
    \]

    Die Seite des Quadrates ist also \(\mathbf{5,4m}\) lang.

  % 15.
  \item Wenn man zwei gegenüber liegende Seiten eines Quadrates um \(3cm\) verlängert,
    so hat das entstehende Rechteck einen um \(4,5m^2\) größeren Flächeninhalt als das Quadrat.
    Wie lange sind die Seiten des ursprünglichen Quadrates?

    \[
      \begin{aligned}
        A_1 &= x^2 \\
        A_2 &= A_1 + 4,5cm^2 = x^2 + 4,5cm^2 \\
        A_2 &= (x + 3cm) \cdot x = x^2 + 3cm \cdot x \\
        x^2 + 3cm \cdot x &= x^2 + 4,5cm^2 &\quad\big| - x^2 \\
        3cm \cdot x &= 4,5cm^2 &\quad\big| : (3cm) \\
        x &= \mathbf{1,5cm}
      \end{aligned}
    \]

    Die Seite des Quadrates ist also \(\mathbf{1,5cm}\) lang.

  % 16.
  \item Wenn man zwei gegenüber liegende Seiten eines Quadrates um \(2cm\) verringert,
    so hat das entstehende Rechteck \(10cm^2\) weniger Flächeninhalt als das Quadrat.
    Wie lange sind die Seiten des ursprünglichen Quadrates?

    \[
      \begin{aligned}
        A_1 &= x^2 \\
        A_2 &= A_1 - 10cm^2 = x^2 - 10cm^2 \\
        A_2 &= (x - 2cm) \cdot x = x^2 - 2cm \cdot x \\
        x^2 - 2cm \cdot x &= x^2 - 10cm^2 &\quad\big| - x^2 \\
        -2cm \cdot x &= -10cm^2 &\quad\big| : (-2cm) \\
        x &= \mathbf{5cm}
      \end{aligned}
    \]

    Die Seite des Quadrates ist also \(\mathbf{5cm}\) lang.

  % 17. 21 Jahre
  \item Tobias ist 28 Jahre alt. Er ist doppelt so alt wie Marc war als Tobias so alt war wie Marc jetzt.
    Wie alt ist Marc? \\
    (Tipp: für diese schwierige Aufgabe: Daten in eine Tabelle eintragen;
    die Gleichung ist: verstrichene Zeit aus Sicht von Marc = verstrichene Zeit aus Sicht von Tobias).

    \[
      \begin{aligned}
        \textnormal{Tobias} &= 28 \\
        \textnormal{Marc} &= x \\
        \textnormal{Verstrichene Zeit (Marc)} &= \textnormal{Verstrichene Zeit (Tobias)} \\
      \end{aligned}
    \]

    Marc ist also \(\mathbf{21}\) Jahre alt.
\end{enumerate}
